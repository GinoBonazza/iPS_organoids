Shared GO-BP pathways

Shared pathways have the same NES direction and FDR < 0.05 in both lines.

GO redundancy is reduced with clusterProfiler::simplify() using a semantic similarity cutoff of 0.6. The representative term within each redundant group is selected using the worse FDR across the two lines, so the simplification does not favour one line.

shared_CM <- inner_join(
  GSEA_CM_JDR_df %>% dplyr::select(ID, Description, NES, pvalue, p.adjust) %>% dplyr::rename(NES_L1 = NES, p_L1 = pvalue, FDR_L1 = p.adjust),
  GSEA_CM_JD6_df %>% dplyr::select(ID, Description, NES, pvalue, p.adjust) %>% dplyr::rename(NES_L2 = NES, p_L2 = pvalue, FDR_L2 = p.adjust),
  by = c("ID", "Description")
) %>%
  filter(sign(NES_L1) == sign(NES_L2), FDR_L1 < 0.05, FDR_L2 < 0.05) %>%
  mutate(worst_FDR = pmax(FDR_L1, FDR_L2), mean_abs_NES = (abs(NES_L1) + abs(NES_L2))/2) %>%
  arrange(worst_FDR, desc(mean_abs_NES))

GSEA_CM_shared_for_simplify <- GSEA_CM_JDR
GSEA_CM_shared_for_simplify@result <- GSEA_CM_JDR@result %>%
  filter(ID %in% shared_CM$ID) %>%
  dplyr::select(-p.adjust) %>%
  left_join(shared_CM %>% dplyr::select(ID, worst_FDR), by = "ID") %>%
  dplyr::rename(p.adjust = worst_FDR)

GSEA_CM_shared_simple <- clusterProfiler::simplify(GSEA_CM_shared_for_simplify, cutoff = 0.6, by = "p.adjust", select_fun = min, measure = "Wang")

Volcano plots

Adjusted P < 0.01 and absolute log2FC > 0.5 are used for colouring, matching the manuscript-style volcano plots.

make_volcano <- function(de, plot_title) {
  keyvals <- rep("grey", nrow(de))
  names(keyvals) <- rep("Not significant", nrow(de))

  keyvals[de$avg_log2FC > 0.5 & de$p_val_adj < 0.01] <- "#A63A2A"
  names(keyvals)[keyvals == "#A63A2A"] <- "Increased"

  keyvals[de$avg_log2FC < -0.5 & de$p_val_adj < 0.01] <- "#004C8C"
  names(keyvals)[keyvals == "#004C8C"] <- "Decreased"

  EnhancedVolcano(de, lab = de$gene, x = "avg_log2FC", y = "p_val_adj", labSize = 0,
                  titleLabSize = 18, subtitleLabSize = 14, axisLabSize = 13, captionLabSize = 0,
                  pointSize = 1.2, FCcutoff = 0.5, pCutoff = 0.01, colCustom = keyvals,
                  colAlpha = 1, drawConnectors = FALSE, boxedLabels = FALSE,
                  subtitle = NULL, title = plot_title) +
    theme(legend.position = "none")
}
